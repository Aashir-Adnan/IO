```dart
import 'package:flutter_secure_storage/flutter_secure_storage.dart';

/// The subset of FlutterSecureStorage's API SecureStore needs — lets tests
/// inject a fake instead of touching the real platform keystore.
abstract class KeyValueStore {
  Future<void> write({required String key, required String? value});
  Future<String?> read({required String key});
}

class PlatformKeyValueStore implements KeyValueStore {
  final FlutterSecureStorage _fs;
  PlatformKeyValueStore([FlutterSecureStorage? fs])
      : _fs = fs ?? const FlutterSecureStorage();

  @override
  Future<void> write({required String key, required String? value}) =>
      value == null ? _fs.delete(key: key) : _fs.write(key: key, value: value);

  @override
  Future<String?> read({required String key}) => _fs.read(key: key);
}

/// The Anthropic API key, in the OS keystore — never plain storage, never
/// logged. See rules/local-first.md and the phone-app-design spec §"API key
/// on-device".
class SecureStore {
  static const _apiKeyName = 'anthropic_api_key';
  final KeyValueStore _kv;
  SecureStore([KeyValueStore? kv]) : _kv = kv ?? PlatformKeyValueStore();

  Future<void> setApiKey(String key) => _kv.write(key: _apiKeyName, value: key);
  Future<String?> getApiKey() => _kv.read(key: _apiKeyName);
  Future<void> clearApiKey() => _kv.write(key: _apiKeyName, value: null);
}
```

```dart
import 'dart:convert';
import 'package:http/http.dart' as http;

class AnthropicApiError implements Exception {
  final int statusCode;
  AnthropicApiError(this.statusCode);
  @override
  String toString() => 'AnthropicApiError($statusCode)';
}

/// The one call onboarding needs: is this key good? A cheap max_tokens:1 turn
/// costs a fraction of a cent and confirms the key works before it's saved.
class AnthropicClient {
  final http.Client http;
  AnthropicClient({required this.http});

  Future<bool> validateKey(String apiKey) async {
    final resp = await http.post(
      Uri.parse('https://api.anthropic.com/v1/messages'),
      headers: {
        'x-api-key': apiKey,
        'anthropic-version': '2023-06-01',
        'content-type': 'application/json',
      },
      body: jsonEncode({
        'model': 'claude-haiku-4-5',
        'max_tokens': 1,
        'messages': [{'role': 'user', 'content': 'hi'}],
      }),
    );
    if (resp.statusCode == 200) return true;
    if (resp.statusCode == 401 || resp.statusCode == 403) return false;
    throw AnthropicApiError(resp.statusCode);
  }
}
```

```dart
import 'package:flutter/material.dart';
import '../../core/anthropic_client.dart';
import '../../core/secure_store.dart';

class ApiKeyScreen extends StatefulWidget {
  final SecureStore store;
  final AnthropicClient client;
  final VoidCallback onDone;
  const ApiKeyScreen(
      {super.key, required this.store, required this.client, required this.onDone});

  @override
  State<ApiKeyScreen> createState() => _ApiKeyScreenState();
}

class _ApiKeyScreenState extends State<ApiKeyScreen> {
  final _controller = TextEditingController();
  bool _checking = false;
  String? _error;

  Future<void> _continue() async {
    final key = _controller.text.trim();
    if (key.isEmpty) return;
    setState(() { _checking = true; _error = null; });
    try {
      final ok = await widget.client.validateKey(key);
      if (!ok) {
        setState(() { _checking = false; _error = "That key didn't work."; });
        return;
      }
      await widget.store.setApiKey(key);
      widget.onDone();
    } catch (_) {
      setState(() { _checking = false; _error = "Couldn't reach Anthropic — try again."; });
    }
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      body: SafeArea(
        child: Padding(
          padding: const EdgeInsets.all(24),
          child: Column(
            mainAxisAlignment: MainAxisAlignment.center,
            crossAxisAlignment: CrossAxisAlignment.stretch,
            children: [
              const Text('Enter your Anthropic API key',
                  style: TextStyle(fontSize: 20, fontWeight: FontWeight.w500)),
              const SizedBox(height: 16),
              TextField(
                controller: _controller,
                obscureText: true,
                decoration: const InputDecoration(hintText: 'sk-ant-…'),
              ),
              if (_error != null) ...[
                const SizedBox(height: 8),
                Text(_error!, style: const TextStyle(color: Colors.redAccent)),
              ],
              const SizedBox(height: 16),
              FilledButton(
                onPressed: _checking ? null : _continue,
                child: Text(_checking ? 'Checking…' : 'Continue'),
              ),
            ],
          ),
        ),
      ),
    );
  }
}
```
